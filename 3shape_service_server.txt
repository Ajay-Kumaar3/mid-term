
#!/usr/bin/env python3

import rospy
import math

from geometry_msgs.msg import Twist
from turtlesim.srv import Spawn, SetPen
from exam_pkg.srv import DrawShapes, DrawShapesResponse

rospy.init_node('shapes_service_server')

# Spawn turtles and set pen colours
rospy.wait_for_service('/spawn')
spawn = rospy.ServiceProxy('/spawn', Spawn)

try:
    spawn(5.5, 6.5, 0.0, 'turtle2')
except rospy.ServiceException:
    rospy.loginfo("turtle2 already exists")

try:
    spawn(5.5, 9.0, 0.0, 'turtle3')
except rospy.ServiceException:
    rospy.loginfo("turtle3 already exists")

rospy.wait_for_service('/turtle1/set_pen')
rospy.wait_for_service('/turtle2/set_pen')
rospy.wait_for_service('/turtle3/set_pen')

set_pen1 = rospy.ServiceProxy('/turtle1/set_pen', SetPen)
set_pen2 = rospy.ServiceProxy('/turtle2/set_pen', SetPen)
set_pen3 = rospy.ServiceProxy('/turtle3/set_pen', SetPen)

set_pen1(255, 0, 0, 3, 0)  # Red
set_pen2(0, 255, 0, 3, 0)  # Green
set_pen3(0, 0, 255, 3, 0)  # Blue

# Publishers
pub1 = rospy.Publisher('/turtle1/cmd_vel', Twist, queue_size=10)
pub2 = rospy.Publisher('/turtle2/cmd_vel', Twist, queue_size=10)
pub3 = rospy.Publisher('/turtle3/cmd_vel', Twist, queue_size=10)

rate = rospy.Rate(50)


def draw_circle(pub):
    speed = 1.0
    radius = 1.0
    ang_speed = speed / radius
    duration = 2 * math.pi / ang_speed

    msg = Twist()
    msg.linear.x = speed
    msg.angular.z = ang_speed

    start = rospy.Time.now()
    while (rospy.Time.now() - start).to_sec() < duration:
        pub.publish(msg)
        rate.sleep()

    pub.publish(Twist())


def draw_diamond(pub):
    forward = Twist()
    forward.linear.x = 1.0

    rotate = Twist()
    rotate.angular.z = -math.pi / 2

    for i in range(4):
        start = rospy.Time.now()
        while (rospy.Time.now() - start).to_sec() < 2.0:
            pub.publish(forward)
            rate.sleep()

        pub.publish(Twist())

        start = rospy.Time.now()
        while (rospy.Time.now() - start).to_sec() < 1.0:
            pub.publish(rotate)
            rate.sleep()

        pub.publish(Twist())


def draw_rectangle(pub):
    forward = Twist()
    forward.linear.x = 1.0

    turn = Twist()
    turn.angular.z = math.pi / 2

    for i in range(2):
        start = rospy.Time.now()
        while (rospy.Time.now() - start).to_sec() < 3.0:
            pub.publish(forward)
            rate.sleep()

        pub.publish(Twist())

        start = rospy.Time.now()
        while (rospy.Time.now() - start).to_sec() < 1.0:
            pub.publish(turn)
            rate.sleep()

        pub.publish(Twist())

        start = rospy.Time.now()
        while (rospy.Time.now() - start).to_sec() < 2.0:
            pub.publish(forward)
            rate.sleep()

        pub.publish(Twist())

        start = rospy.Time.now()
        while (rospy.Time.now() - start).to_sec() < 1.0:
            pub.publish(turn)
            rate.sleep()

        pub.publish(Twist())


def handle_draw_shapes(request):
    rospy.loginfo("Drawing circle")
    draw_circle(pub1)

    rospy.loginfo("Drawing diamond")
    draw_diamond(pub2)

    rospy.loginfo("Drawing rectangle")
    draw_rectangle(pub3)

    response = DrawShapesResponse()
    response.success = True
    response.message = "All three shapes completed"

    return response


server = rospy.Service(
    '/draw_shapes_service',
    DrawShapes,
    handle_draw_shapes
)

rospy.loginfo("Draw Shapes Service Server is ready")
rospy.spin()

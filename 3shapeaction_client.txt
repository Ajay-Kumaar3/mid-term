#!/usr/bin/env python3

import rospy
import actionlib

from exam_pkg.msg import DrawShapesAction
from exam_pkg.msg import DrawShapesGoal

rospy.init_node('draw_shape_client')

def feedback_callback(feedback):
    print("Current shape:", feedback.current_shape)
    print("Shapes completed:", feedback.shapes_completed)
    	
client = actionlib.SimpleActionClient('/draw_shapes',DrawShapesAction)

client.wait_for_server()

goal = DrawShapesGoal()

goal.shapes_to_draw = 3

client.send_goal(goal,feedback_cb = feedback_callback)

client.wait_for_result()

result = client.get_result()

print("Success:", result.success)
print("Message:", result.message)
